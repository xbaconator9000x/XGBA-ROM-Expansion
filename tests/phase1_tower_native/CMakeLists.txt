cmake_minimum_required(VERSION 3.16)
project(XGBA_Phase1_Tower_Native_mGBA_Flash C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(BUILD_QT OFF CACHE BOOL "" FORCE)
set(BUILD_SDL OFF CACHE BOOL "" FORCE)
set(BUILD_LIBRETRO OFF CACHE BOOL "" FORCE)
set(BUILD_TEST OFF CACHE BOOL "" FORCE)
set(BUILD_STATIC ON CACHE BOOL "" FORCE)
set(BUILD_SHARED OFF CACHE BOOL "" FORCE)
set(BUILD_HEADLESS OFF CACHE BOOL "" FORCE)
set(ENABLE_SCRIPTING OFF CACHE BOOL "" FORCE)
set(BUILD_PYTHON OFF CACHE BOOL "" FORCE)
set(USE_PNG OFF CACHE BOOL "" FORCE)
set(USE_ZLIB OFF CACHE BOOL "" FORCE)
set(ENABLE_GDB_STUB OFF CACHE BOOL "" FORCE)
set(MGBA_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../upstream-mgba")
if(NOT EXISTS "${MGBA_ROOT}/src/gba/savedata.c")
  message(FATAL_ERROR "Pinned mGBA source checkout is required")
endif()
add_subdirectory("${MGBA_ROOT}" "${CMAKE_BINARY_DIR}/pinned-mgba")
add_executable(phase1_tower_mgba_flash "${CMAKE_CURRENT_SOURCE_DIR}/phase1_tower_mgba_flash.c")
target_link_libraries(phase1_tower_mgba_flash PRIVATE mgba)
get_target_property(_mgba_defs mgba COMPILE_DEFINITIONS)
if(NOT _mgba_defs)
  message(FATAL_ERROR "Missing original mGBA compiler ABI definitions")
endif()
target_compile_definitions(phase1_tower_mgba_flash PRIVATE ${_mgba_defs})
target_include_directories(phase1_tower_mgba_flash PRIVATE
  "${MGBA_ROOT}/include" "${CMAKE_BINARY_DIR}/pinned-mgba/include")

# Test-only direct entry into unchanged original-agbcc linked FireRed stub.
add_executable(phase1_tower_original_arm_stub "${CMAKE_CURRENT_SOURCE_DIR}/phase1_tower_original_arm_stub.c")
target_link_libraries(phase1_tower_original_arm_stub PRIVATE mgba)
target_compile_definitions(phase1_tower_original_arm_stub PRIVATE ${_mgba_defs})
target_include_directories(phase1_tower_original_arm_stub PRIVATE
  "${MGBA_ROOT}/include" "${CMAKE_BINARY_DIR}/pinned-mgba/include")
