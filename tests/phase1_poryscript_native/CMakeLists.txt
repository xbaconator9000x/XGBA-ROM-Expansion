cmake_minimum_required(VERSION 3.16)
project(XGBA_Phase1_Original_AGBCC_Reader_Native_mGBA C)
set(CMAKE_C_STANDARD 11)
set(BUILD_QT OFF CACHE BOOL "" FORCE)
set(BUILD_SDL OFF CACHE BOOL "" FORCE)
set(BUILD_LIBRETRO OFF CACHE BOOL "" FORCE)
set(BUILD_TEST OFF CACHE BOOL "" FORCE)
set(BUILD_STATIC ON CACHE BOOL "" FORCE)
set(BUILD_SHARED OFF CACHE BOOL "" FORCE)
set(BUILD_HEADLESS OFF CACHE BOOL "" FORCE)
set(ENABLE_SCRIPTING OFF CACHE BOOL "" FORCE)
set(BUILD_PYTHON OFF CACHE BOOL "" FORCE)
set(USE_PNG OFF CACHE BOOL "" FORCE)
set(USE_ZLIB OFF CACHE BOOL "" FORCE)
set(ENABLE_GDB_STUB OFF CACHE BOOL "" FORCE)
set(MGBA_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../upstream-mgba")
if(NOT EXISTS "${MGBA_ROOT}/src/gba/memory.c")
  message(FATAL_ERROR "Pinned mGBA checkout required")
endif()
add_subdirectory("${MGBA_ROOT}" "${CMAKE_BINARY_DIR}/pinned-mgba")
add_executable(phase1_original_reader_native "${CMAKE_CURRENT_SOURCE_DIR}/reader.c")
target_link_libraries(phase1_original_reader_native PRIVATE mgba)
get_target_property(_defs mgba COMPILE_DEFINITIONS)
if(NOT _defs)
  message(FATAL_ERROR "Original mGBA compiler definitions missing")
endif()
target_compile_definitions(phase1_original_reader_native PRIVATE ${_defs})
target_include_directories(phase1_original_reader_native PRIVATE
  "${MGBA_ROOT}/include" "${CMAKE_BINARY_DIR}/pinned-mgba/include")
